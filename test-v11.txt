```ts
import fs from 'fs';
import path from 'path';
import ignore from 'ignore';

export interface ScanOptions {
    includeExts?: string[];
    sinceMs?: number;
}

/**
 * Basic heuristic to check if file is text or binary
 * Read first 512 bytes and check for null bytes
 */
function isBinaryFileSync(filePath: string): boolean {
    const buffer = Buffer.alloc(512);
    try {
        const fd = fs.openSync(filePath, 'r');
        const bytesRead = fs.readSync(fd, buffer, 0, 512, 0);
        fs.closeSync(fd);
        for (let i = 0; i < bytesRead; i++) {
            if (buffer[i] === 0) return true;
        }
    } catch (e) {
        return true; // Ignore errors, assume binary to be safe
    }
    return false;
}

function getIgnorer(dirPath: string) {
    const ig = ignore();
    ig.add(['.git', 'node_modules', 'dist', 'build', '.env', '*.log', '.DS_Store']);
    const gitignorePath = path.join(dirPath, '.gitignore');
    if (fs.existsSync(gitignorePath)) {
        ig.add(fs.readFileSync(gitignorePath, 'utf8'));
    }
    return ig;
}

export function scanDirectory(dirPath: string, rootPath = dirPath, ig = getIgnorer(rootPath), options: ScanOptions = {}): string[] {
    let results: string[] = [];
    if (!fs.existsSync(dirPath)) return results;

    const list = fs.readdirSync(dirPath);
    for (const file of list) {
        const fullPath = path.join(dirPath, file);
        const relativePath = path.relative(rootPath, fullPath).replace(/\\/g, '/');

        if (ig.ignores(relativePath)) continue;

        const stat = fs.statSync(fullPath);
        if (stat.isDirectory()) {
            results = results.concat(scanDirectory(fullPath, rootPath, ig, options));
        } else {
            // Check extension
            if (options.includeExts && options.includeExts.length > 0) {
                const ext = path.extname(file);
                if (!options.includeExts.includes(ext) && !options.includeExts.includes(file)) {
                    continue;
                }
            }
            // Check modified date
            if (options.sinceMs && stat.mtimeMs < options.sinceMs) {
                continue;
            }
            results.push(relativePath);
        }
    }
    return results;
}

export function generateTree(files: string[]): string {
    const tree: any = {};
    for (const file of files) {
        const parts = file.split('/');
        let current = tree;
        for (let i = 0; i < parts.length; i++) {
            const part = parts[i];
            if (!current[part]) {
                current[part] = i === parts.length - 1 ? null : {};
            }
            current = current[part];
        }
    }

    function buildString(node: any, prefix = ''): string {
        let result = '';
        const keys = Object.keys(node);
        for (let i = 0; i < keys.length; i++) {
            const key = keys[i];
            const isLast = i === keys.length - 1;
            result += `${prefix}${isLast ? '└── ' : '├── '}${key}\n`;
            if (node[key] !== null) {
                result += buildString(node[key], prefix + (isLast ? '    ' : '│   '));
            }
        }
        return result;
    }
    return buildString(tree);
}

export function readFilesContent(rootPath: string, files: string[]): { [key: string]: string } {
    const contents: { [key: string]: string } = {};
    for (const file of files) {
        const fullPath = path.join(rootPath, file);
        try {
            const stat = fs.statSync(fullPath);
            if (stat.size > 1024 * 1024) {
                contents[file] = `// [Ignored: > 1MB]`;
                continue;
            }
            if (isBinaryFileSync(fullPath)) {
                contents[file] = `// [Ignored: Binary File]`;
                continue;
            }
            contents[file] = fs.readFileSync(fullPath, 'utf8');
        } catch (e) {
            contents[file] = `// [Error reading file]`;
        }
    }
    return contents;
}
```

```ts
export type TargetModel = 'gpt' | 'claude';

export interface FormatOptions {
    tree: string;
    contents: { [path: string]: string };
    target: TargetModel;
    issue?: string;
    focus?: string[];
}

export function formatContext(options: FormatOptions): string {
    let result = '';

    // Si on a un focus, on le met en évidence
    const focusFiles = options.focus || [];
    const normalFiles = Object.keys(options.contents).filter(f => !focusFiles.includes(f));
    const focusedFilesList = Object.keys(options.contents).filter(f => focusFiles.includes(f));

    if (options.target === 'claude') {
        if (options.issue) {
            result += `<user_instruction>\n${options.issue}\n</user_instruction>\n\n`;
        }

        result += `<project_context>\n`;
        result += `  <project_tree>\n${options.tree}\n  </project_tree>\n`;
        
        result += `  <project_files>\n`;

        // Inject focused files first with a special tag
        for (const filePath of focusedFilesList) {
            result += `    <file path="${filePath}" focus="true">\n`;
            result += `${options.contents[filePath]}\n`;
            result += `    </file>\n`;
        }

        for (const filePath of normalFiles) {
            result += `    <file path="${filePath}">\n`;
            result += `${options.contents[filePath]}\n`;
            result += `    </file>\n`;
        }
        
        result += `  </project_files>\n`;
        result += `</project_context>`;
    } else {
        // GPT / Markdown
        if (options.issue) {
            result += `# User Instruction\n\n**${options.issue}**\n\n---\n\n`;
        }

        result += `# Project Context\n\n`;
        result += `## Directory Structure\n`;
        result += `\`\`\`text\n${options.tree}\n\`\`\`\n\n`;
        
        result += `## Files\n\n`;

        if (focusedFilesList.length > 0) {
            result += `### 🔥 FOCUSED FILES\n\n`;
            for (const filePath of focusedFilesList) {
                result += `#### \`${filePath}\`\n`;
                const ext = filePath.split('.').pop() || 'text';
                result += `\`\`\`${ext}\n${options.contents[filePath]}\n\`\`\`\n\n`;
            }
            result += `### Other Files\n\n`;
        }

        for (const filePath of normalFiles) {
            result += `#### \`${filePath}\`\n`;
            const ext = filePath.split('.').pop() || 'text';
            result += `\`\`\`${ext}\n${options.contents[filePath]}\n\`\`\`\n\n`;
        }
    }

    return result;
}
```

#### `src/index.ts`
```ts
#!/usr/bin/env node
import { Command } from 'commander';
import { scanDirectory, generateTree, readFilesContent } from './scanner';
import { countTokens } from './tokenCounter';
import { formatContext, TargetModel } from './formatter';
import { showBanner, showHelp, showStep, showSuccess, showWarning, showInfo, showTokenCount } from './ui';
import { runInteractiveMode } from './interactive';
import clipboardy from 'clipboardy';
import fs from 'fs';
import path from 'path';

const program = new Command();

program
    .name('code-caricature')
    .description('🎨 Fais la caricature de ton code, donne-la à ton IA !')
    .version('1.1.0');

// ─── MODE RAPIDE (Expert) ───────────────────────────────────────────────

program
    .command('export')
    .description('Exporter le contexte du projet (mode rapide)')
    .option('-t, --target <model>', 'Modèle cible : gpt ou claude', 'gpt')
    .option('-o, --output <file>', 'Sauvegarder dans un fichier')
    .option('-i, --include <exts>', 'Extensions (ex: .ts,.js)')
    .option('-s, --since <hours>', 'Fichiers modifiés depuis N heures')
    .option('-f, --focus <file>', 'Fichier à mettre en évidence')
    .option('-q, --issue <text>', 'Question/problème à injecter')
    .action((options) => {
        showBanner();

        const targetDir = process.cwd();
        showStep('🔍', `Scan : ${targetDir}`);

        let includeExts: string[] = [];
        if (options.include) {
            includeExts = options.include.split(',').map((e: string) => e.trim());
        }

        let sinceMs: number | undefined;
        if (options.since) {
            const hours = parseFloat(options.since);
            sinceMs = Date.now() - (hours * 60 * 60 * 1000);
            showStep('🕐', `Filtre : modifiés dans les ${hours} dernières heures`);
        }

        const files = scanDirectory(targetDir, targetDir, undefined, { includeExts, sinceMs });
        showStep('📁', `${files.length} fichiers trouvés`);

        const tree = generateTree(files);
        const contents = readFilesContent(targetDir, files);

        const targetModel: TargetModel = (options.target?.toLowerCase() === 'claude') ? 'claude' : 'gpt';
        const focusFiles = options.focus ? [options.focus] : [];

        showStep('🎨', 'Création de la caricature...');

        const formatted = formatContext({
            tree,
            contents,
            target: targetModel,
            issue: options.issue,
            focus: focusFiles,
        });

        const tokens = countTokens(formatted);
        showTokenCount(tokens);

        if (options.output) {
            const outPath = path.resolve(targetDir, options.output);
            fs.writeFileSync(outPath, formatted, 'utf8');
            showSuccess(`Fichier créé : ${outPath}`);
            showInfo('Glissez-déposez ce fichier dans ChatGPT ou Claude.');
        } else {
            clipboardy.writeSync(formatted);
            showSuccess('Caricature copiée dans le presse-papiers !');
            showInfo('Allez sur ChatGPT/Claude et appuyez sur Ctrl+V.');
        }
    });

// ─── COMMANDE HELP ──────────────────────────────────────────────────────

program
    .command('help')
    .description('Afficher le guide complet avec toutes les commandes')
    .action(() => {
        showBanner();
        showHelp();
    });

// ─── MODE INTERACTIF (par défaut, sans arguments) ───────────────────────

// If no command is given, launch interactive mode
if (process.argv.length <= 2) {
    runInteractiveMode().catch(console.error);
} else {
    program.parse(process.argv);
}
```

```ts
/* eslint-disable @typescript-eslint/no-var-requires */
const chalk = require('chalk');
import { select, input, confirm, checkbox } from '@inquirer/prompts';
import fs from 'fs';
import path from 'path';
import { scanDirectory, generateTree, readFilesContent } from './scanner';
import { countTokens } from './tokenCounter';
import { formatContext, TargetModel } from './formatter';
import clipboardy from 'clipboardy';
import { showBanner, showStep, showSuccess, showInfo, showTokenCount } from './ui';

/**
 * The interactive menu mode - guides the user step by step
 */
export async function runInteractiveMode(): Promise<void> {
    showBanner();
    console.log(chalk.gray('  Bienvenue ! Ce menu va vous guider pas à pas.\n'));

    // Step 1: Choose action
    const action = await select({
        message: '🎯  Que voulez-vous faire ?',
        choices: [
            {
                name: '📋  Exporter tout mon projet',
                value: 'full',
                description: 'Scan complet du dossier actuel'
            },
            {
                name: '🔍  Filtrer par types de fichiers',
                value: 'filter',
                description: 'Seulement .ts, .js, .py, etc.'
            },
            {
                name: '🕐  Fichiers modifiés récemment',
                value: 'recent',
                description: 'Seulement ce qui a changé'
            },
            {
                name: '📖  Voir l\'aide complète',
                value: 'help',
                description: 'Explications de toutes les commandes'
            },
        ],
    });

    if (action === 'help') {
        const { showHelp } = await import('./ui');
        showHelp();
        return;
    }

    // Step 2: Get scan options
    let includeExts: string[] = [];
    let sinceMs: number | undefined;

    if (action === 'filter') {
        const extsInput = await input({
            message: '📝  Quelles extensions ? (séparées par des virgules)',
            default: '.ts,.js',
        });
        includeExts = extsInput.split(',').map(e => e.trim());
    }

    if (action === 'recent') {
        const hoursInput = await input({
            message: '🕐  Depuis combien d\'heures ?',
            default: '24',
        });
        const hours = parseFloat(hoursInput);
        sinceMs = Date.now() - (hours * 60 * 60 * 1000);
    }

    // Step 3: Focus file?
    const wantFocus = await confirm({
        message: '🔥  Voulez-vous mettre un fichier en évidence (Focus) ?',
        default: false,
    });

    let focusFiles: string[] = [];
    if (wantFocus) {
        const targetDir = process.cwd();
        const allFiles = scanDirectory(targetDir, targetDir, undefined, {});
        
        if (allFiles.length > 0) {
            const selected = await checkbox({
                message: '📂  Sélectionnez les fichiers à mettre en évidence :',
                choices: allFiles.slice(0, 30).map(f => ({
                    name: f,
                    value: f,
                })),
            });
            focusFiles = selected;
        }
    }

    // Step 4: Inject issue?
    const wantIssue = await confirm({
        message: '❓  Voulez-vous ajouter une question/problème pour l\'IA ?',
        default: false,
    });

    let issue: string | undefined;
    if (wantIssue) {
        issue = await input({
            message: '💬  Décrivez votre problème ou question :',
        });
    }

    // Step 5: Target model
    const targetModel = await select({
        message: '🤖  Quelle IA allez-vous utiliser ?',
        choices: [
            { name: '🟢  ChatGPT / GPT (Markdown)', value: 'gpt' as TargetModel },
            { name: '🟣  Claude (XML)', value: 'claude' as TargetModel },
        ],
    });

    // Step 6: Output mode
    const outputMode = await select({
        message: '📤  Comment voulez-vous récupérer le résultat ?',
        choices: [
            {
                name: '📋  Copier dans le presse-papiers (Ctrl+V ensuite)',
                value: 'clipboard',
            },
            {
                name: '📁  Créer un fichier texte (glisser-déposer)',
                value: 'file',
            },
        ],
    });

    let outputFile: string | undefined;
    if (outputMode === 'file') {
        outputFile = await input({
            message: '📝  Nom du fichier à créer :',
            default: 'code-caricature.txt',
        });
    }

    // Execute!
    console.log('');
    console.log(chalk.gray('  ─────────────────────────────────────────────────────────'));
    
    const targetDir = process.cwd();
    showStep('🔍', `Scan du dossier : ${chalk.cyan(targetDir)}`);

    const files = scanDirectory(targetDir, targetDir, undefined, { includeExts, sinceMs });
    showStep('📁', `${chalk.bold(files.length.toString())} fichiers trouvés`);

    const tree = generateTree(files);
    const contents = readFilesContent(targetDir, files);

    showStep('🎨', 'Création de la caricature...');

    const formatted = formatContext({
        tree,
        contents,
        target: targetModel,
        issue,
        focus: focusFiles,
    });

    const tokens = countTokens(formatted);
    showTokenCount(tokens);

    if (outputFile) {
        const outPath = path.resolve(targetDir, outputFile);
        fs.writeFileSync(outPath, formatted, 'utf8');
        showSuccess(`Fichier créé : ${outPath}`);
        showInfo('Glissez-déposez ce fichier dans ChatGPT ou Claude.');
    } else {
        clipboardy.writeSync(formatted);
        showSuccess('Caricature copiée dans le presse-papiers !');
        showInfo('Allez sur ChatGPT/Claude et appuyez sur Ctrl+V.');
    }

    console.log(chalk.gray('  ─────────────────────────────────────────────────────────\n'));
}
```

#### `src/tokenCounter.ts`
```ts
import { get_encoding } from 'tiktoken';

/**
 * Compte le nombre de tokens d'un texte pour un modèle donné
 * (par défaut gpt-4o, qui utilise o200k_base ou cl100k_base)
 */
export function countTokens(text: string): number {
    // cl100k_base est utilisé par GPT-4, GPT-3.5-turbo, etc.
    const enc = get_encoding("cl100k_base");
    try {
        const tokens = enc.encode(text);
        return tokens.length;
    } finally {
        enc.free(); // important pour libérer la mémoire WASM
    }
}
```

```ts
/**
 * Show token count with color coding
 */
export function showTokenCount(tokens: number): void {
    let color = chalk.green;
    let label = 'OK';
    if (tokens > 100000) {
        color = chalk.red;
        label = 'TRÈS ÉLEVÉ';
    } else if (tokens > 50000) {
        color = chalk.yellow;
        label = 'ÉLEVÉ';
    } else if (tokens > 20000) {
        color = chalk.cyan;
        label = 'MOYEN';
    }

    console.log(color(`  📊  Tokens estimés : ${tokens.toLocaleString()} [${label}]`));
    
    if (tokens > 100000) {
        showWarning('Ce volume pourrait dépasser les limites de certaines IA.');
        console.log(chalk.gray('     💡 Astuce : Utilisez --include ou --focus pour réduire la taille.'));
    }
}
```
